while world.fights():
    teach(love)

    if fear.spreads():
        plant(trust)

    if voices.break():
        listen(deeper)

    if anger.rises():
        lower(ego)

    if distance.grows():
        build(bridges)

    for wound in humanity:
        care(wound)

    for child in tomorrow:
        teach(child, kindness)

    if hate.repeats():
        refuse(to_become=hate)

    hope += 1
    mercy += 1
    courage += 1

    world = world.try_again()
